# Recipe 2: layer two sounds in one part, then split them with LEARN, a key range and an octave shift.
# (midi bytes are hex: 3c = middle C4, 53 = B5; the screen uses standard naming, C4 = 60)
boot
wipe
run 400
# part 8 makes room (LFO), part 1 gets a second layer (ENV): both layers play every key
enc 0 -1
tap btn 5
expect_part 7 0 0
tap btn 4
expect_part 0 1 2
tap btn 11
shot 2a-two-layers
# layered: one key reaches both engines
midi 1 90 3c 64
run 80
expect_voices 2
midi 1 80 3c 00
run 80
expect_voices 0
# LEARN: play the lowest then the highest key; the framed layer (layer 2, on engine 8) takes that range
tap btn 10
shot 2b-learn-armed
midi 1 90 3c 64
run 60
midi 1 80 3c 00
run 60
midi 1 90 53 64
run 60
midi 1 80 53 00
run 120
expect_engine 7 60 83
shot 2c-range-learned
# the Range page: K3 shifts the layer up two octaves
tap btn 6
tap btn 6
expect_screen 4
enc 4 2
expect_engine 7 60 83
shot 2d-shifted-two-octaves
# ALGO frames layer 1, K2 gives it the lower half only: high = B3 (59)
enc 1 -1
enc 3 -68
expect_engine 0 0 59
shot 2e-first-layer-below
# now a key in the lower half reaches layer 1 only, a key in the upper half layer 2 only
midi 1 90 30 64
run 80
expect_voices 1
midi 1 90 40 64
run 80
expect_voices 2
midi 1 80 30 00
midi 1 80 40 00
run 80
expect_voices 0
tap btn 8
shot 2f-parts-show-the-split
